    \"\"\"Circular convolution via 1D Fast Fourier Transform.\"\"\"

    history = []
    H = torch.zeros(self.dim)
    alarms = []

    for t, a_t in enumerate(trace_vectors):
        if t == 0:
            H = a_t
        else:
            rotated_H = torch.matmul(self.R, H)
            H = self._circ_conv(rotated_H, a_t)

            sim = torch.cosine_similarity(H.unsqueeze(0), prev_H.unsqueeze(0)).item()

Measures cosine recovery after soft-pruning high-variance noise coordinates (Lane L-01).
"""

    f_noisy = noisy_embeddings * mask

    n_noisy = torch.nn.functional.normalize(f_noisy, dim=1)

    cosine_rec = torch.sum(n_clean * n_noisy, dim=1).mean().item()
